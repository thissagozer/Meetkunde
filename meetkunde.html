const XMIN=-2,XMAX=10,YMIN=-4,YMAX=8,S=45,W=(XMAX-XMIN)*S,H=(YMAX-YMIN)*S;

function clipT(g,m=0){

  let [t0,t1]=domain(g);

  for(const [dd,pp,lo,hi] of [[g.d.x,g.p.x,XMIN-m,XMAX+m],[g.d.y,g.p.y,YMIN-m,YMAX+m]]){

    if(Math.abs(dd)<1e-12){ if(pp<lo||pp>hi) return null; }

    else{ let a=(lo-pp)/dd,b=(hi-pp)/dd; if(a>b)[a,b]=[b,a]; t0=Math.max(t0,a); t1=Math.min(t1,b); }

  }

  return t0<t1?[t0,t1]:null;

}

function lineCircleTs(g,c,r){

  const f=sub(g.p,c),a=dot(g.d,g.d),b=2*dot(g.d,f),cc=dot(f,f)-r*r,D=b*b-4*a*cc;

  if(D<0) return []; if(D<1e-12) return [-b/(2*a)];

  const s=Math.sqrt(D); return [(-b-s)/(2*a),(-b+s)/(2*a)];

}

function intersect(g1,g2){

  const L1=g1.k!=='circle',L2=g2.k!=='circle',res=[];

  if(L1&&L2){

    const den=cross(g1.d,g2.d); if(Math.abs(den)<1e-12) return res;

    const w=sub(g2.p,g1.p),t=cross(w,g2.d)/den,s=cross(w,g1.d)/den;

    if(inDom(g1,t)&&inDom(g2,s)) res.push({P:at(g1,t),t1:t,t2:s});

  }else if(L1&&!L2){

    for(const t of lineCircleTs(g1,g2.c,g2.r)) if(inDom(g1,t)){const P=at(g1,t);res.push({P,t1:t,t2:ang(g2,P)});}

  }else if(!L1&&L2){

    return intersect(g2,g1).map(o=>({P:o.P,t1:o.t2,t2:o.t1}));

  }else{

    const d=dist(g1.c,g2.c);

    if(d<1e-9||d>g1.r+g2.r+1e-9||d<Math.abs(g1.r-g2.r)-1e-9) return res;

    const a=(g1.r*g1.r-g2.r*g2.r+d*d)/(2*d),h=Math.sqrt(Math.max(0,g1.r*g1.r-a*a));

    const u=mul(sub(g2.c,g1.c),1/d),base=add(g1.c,mul(u,a)),v=perp(u);

    const Ps=h<1e-9?[base]:[add(base,mul(v,h)),sub(base,mul(v,h))];

    for(const P of Ps) res.push({P,t1:ang(g1,P),t2:ang(g2,P)});

  }

  return res;

}

function distToGeom(X,g){

  if(g.k==='circle') return Math.abs(dist(X,g.c)-g.r);

  const [lo,hi]=domain(g); let t=dot(sub(X,g.p),g.d)/dot(g.d,g.d); t=Math.max(lo,Math.min(hi,t));

  return dist(X,at(g,t));

}

function distToPiece(X,pc){

  const g=pc.g;

  if(g.k==='circle'){

    let th=ang(g,X); while(th<pc.a) th+=TAU;

    if(th<=pc.b) return Math.abs(dist(X,g.c)-g.r);

    return Math.min(dist(X,at(g,pc.a)),dist(X,at(g,pc.b)));

  }

  let t=dot(sub(X,g.p),g.d)/dot(g.d,g.d); t=Math.max(pc.a,Math.min(pc.b,t));

  return dist(X,at(g,t));

}

function samplePiece(pc,step=.05){

  const g=pc.g,L=g.k==='circle'?(pc.b-pc.a)*g.r:(pc.b-pc.a)*len(g.d);

  const n=Math.max(2,Math.ceil(L/step)),out=[];

  for(let i=0;i<=n;i++) out.push(at(g,pc.a+(pc.b-pc.a)*i/n));

  return out;

}

function sampleGeom(g){

  if(g.k==='circle') return samplePiece({g,a:0,b:TAU}).filter(inView);

  const r=clipT(g); return r?samplePiece({g,a:r[0],b:r[1]}):[];

}

function condMargin(c,X){

  if(c.t==='in') return c.r-dist(X,c.c);

  if(c.t==='out') return dist(X,c.c)-c.r;

  return c.s*cross(c.d,sub(X,c.p))/len(c.d);

}

    case 'evenw': return f.names[0]===s.names[0]&&same(f.names.slice(1),s.names.slice(1));

    case 'afstand': return same(f.names,s.names)&&Math.abs(f.val-s.val)<1e-6;

  }

  return false;

}

function genMllDrie(){

function genBissCirkel(){

  const D=rp(); if(!farApart([A,B,C,D],2)) return null;

function buildPieces(){

  pieces=[];ipoints=[];

  const G=[]; figs.forEach(f=>f.geoms.forEach((g,i)=>G.push({g,id:f.id+'_'+i})));

  const cuts=G.map(()=>[]);

  for(let i=0;i<G.length;i++) for(let j=i+1;j<G.length;j++)

    for(const o of intersect(G[i].g,G[j].g)){

      cuts[i].push(o.t1);cuts[j].push(o.t2);

      if(inView(o.P)&&!ipoints.some(P=>dist(P,o.P)<1e-6)) ipoints.push(o.P);

    }

  G.forEach((E,i)=>{

    const g=E.g;

    if(g.k==='circle'){

      let a=cuts[i].map(normAng).sort((x,y)=>x-y).filter((v,k,arr)=>k===0||v-arr[k-1]>1e-7);

      if(a.length>1&&a[0]+TAU-a[a.length-1]<1e-7) a.pop();

      if(!a.length) pieces.push({gid:E.id,g,a:0,b:TAU});

      else for(let k=0;k<a.length;k++) pieces.push({gid:E.id,g,a:a[k],b:k<a.length-1?a[k+1]:a[0]+TAU});

    }else{

      const r=clipT(g); if(!r) return;

      const ts=[r[0],...cuts[i].filter(t=>t>r[0]+1e-9&&t<r[1]-1e-9),r[1]].sort((x,y)=>x-y).filter((v,k,arr)=>k===0||v-arr[k-1]>1e-7);

      for(let k=0;k<ts.length-1;k++) pieces.push({gid:E.id,g,a:ts[k],b:ts[k+1]});

    }

  });

}

const pieceKey=pc=>pc.gid+':'+pc.a.toFixed(5);

function regionAt(X){

  const conds=[],key=[]; let edge=false;

  for(const f of figs) f.geoms.forEach((g,i)=>{

    if(g.k==='line'){

      const v=cross(g.d,sub(X,g.p))/len(g.d); if(Math.abs(v)<.05){edge=true;return;}

      conds.push({t:'side',p:g.p,d:g.d,s:Math.sign(v)}); key.push(f.id+i+(v>0?'+':'-'));

    }else if(g.k==='circle'){

      const v=dist(X,g.c)-g.r; if(Math.abs(v)<.05){edge=true;return;}

      conds.push({t:v<0?'in':'out',c:g.c,r:g.r}); key.push(f.id+i+(v<0?'i':'o'));

    }

  });

  if(edge) return 'edge'; if(!conds.length) return null;

  return {conds,key:key.join('|')};

}

function clipShape(c){

  if(c.t==='in') return `<circle cx="${px(c.c.x)}" cy="${py(c.c.y)}" r="${c.r*S}"/>`;

  if(c.t==='out'){const R=c.r*S,x=px(c.c.x),y=py(c.c.y);

    return `<path clip-rule="evenodd" d="M-10,-10 H${W+10} V${H+10} H-10 Z M${x-R},${y} a${R},${R} 0 1,0 ${2*R},0 a${R},${R} 0 1,0 ${-2*R},0 Z"/>`;}

  const u=norm(c.d); let n=perp(u); if(c.s<0) n=mul(n,-1); const L=100;

  const p1=add(c.p,mul(u,L)),p2=add(p1,mul(n,L)),p4=sub(c.p,mul(u,L)),p3=add(p4,mul(n,L));

  return `<polygon points="${[p1,p2,p3,p4].map(P=>px(P.x)+','+py(P.y)).join(' ')}"/>`;

}

function clipped(content,conds,defs){

  let s=content;

  for(const c of conds){const id='cp'+(clipN++); defs.push(`<clipPath id="${id}">${clipShape(c)}</clipPath>`); s=`<g clip-path="url(#${id})">${s}</g>`;}

  return s;

}

function geomSVG(g,color,w,dash=''){

  const da=dash?` stroke-dasharray="${dash}"`:'';

  if(g.k==='circle') return `<circle cx="${px(g.c.x)}" cy="${py(g.c.y)}" r="${g.r*S}" fill="none" stroke="${color}" stroke-width="${w}"${da}/>`;

  const r=clipT(g,1); if(!r) return '';

  const P=at(g,r[0]),Q=at(g,r[1]);

  return `<line x1="${px(P.x)}" y1="${py(P.y)}" x2="${px(Q.x)}" y2="${py(Q.y)}" stroke="${color}" stroke-width="${w}" stroke-linecap="round"${da}/>`;

}

});

function evalAnswer(){

  const A=cur.answer,hp=selPieces.length,hr=selRegions.length,ht=selPoints.length;

  if(!hp&&!hr&&!ht) return {ok:false,msg:'Je hebt nog niets rood gekleurd. Kies 🖍️ Kleuren en klik op het juiste deel.'};

  if(A.kind==='point'){

    if(hp||hr) return {ok:false,msg:'Het antwoord is geen lijn en geen gebied, maar één punt. Kleur alleen dat punt.'};

    const ok=A.pts.every(P=>selPoints.some(Q=>dist(P,Q)<.1))&&selPoints.every(Q=>A.pts.some(P=>dist(P,Q)<.1));

    return ok?{ok:true}:{ok:false,msg:'Dit is niet het juiste punt. Kijk welke twee lijnen elkaar moeten snijpen.'};

  }

  if(A.kind==='curve'){

    if(hr) return {ok:false,msg:'Het antwoord is geen gebied, maar een deel van een lijn of cirkel. Wis de kleuren en klik op de juiste stukken.'};

    if(ht&&!hp) return {ok:false,msg:'Het antwoord bestaat niet uit losse punten, maar uit een deel van een lijn of cirkel.'};

    let tot=0,cov=0,stot=0,sok=0;

    for(const part of A.parts) for(const X of sampleGeom(part.g))

      if(part.conds.every(c=>condMargin(c,X)>.1)){tot++; if(selPieces.some(pc=>distToPiece(X,pc)<.1)) cov++;}

    for(const pc of selPieces) for(const X of samplePiece(pc)){

      if(!inView(X)) continue; stot++;

      if(A.parts.some(part=>distToGeom(X,part.g)<.1&&part.conds.every(c=>condMargin(c,X)>-.1))) sok++;

    }

    const covOk=!tot||cov/tot>.95,precOk=!stot||sok/stot>.95;

    if(covOk&&precOk&&!ht) return {ok:true};

    if(!precOk) return {ok:false,msg:'Je hebt ook punten rood gekleurd die niet aan alle voorwaarden voldoen.'};

    if(ht) return {ok:false,msg:'Losse punten horen hier niet bij het antwoord. Klik ze weg.'};

    return {ok:false,msg:'Wat je gekleurd hebt klopt, maar je mist nog een deel. Kijk of er nog meer stukken zijn die bij het antwoord horen.'};

  }

  if(!hr) return {ok:false,msg:'Het antwoord is een gebied. Klik binnen het gebied om het te kleuren.'};

  let corr=0,miss=0,extra=0;

  for(let x=XMIN+.125;x<XMAX;x+=.25) for(let y=YMIN+.125;y<YMAX;y+=.25){

    const X=V(x,y);

    if(A.parts.some(p=>p.conds.some(c=>Math.abs(condMargin(c,X))<.1))) continue;

    const inC=A.parts.some(p=>p.conds.every(c=>condMargin(c,X)>0));

    const inS=selRegions.some(r=>r.conds.every(c=>condMargin(c,X)>0));

    if(inC) corr++; if(inC&&!inS) miss++; if(!inC&&inS) extra++;

  }

  if((miss+extra)/Math.max(corr,1)<.05) return {ok:true};

  if(extra) return {ok:false,msg:'Je hebt ook een gebied gekleurd dat niet aan alle voorwaarden voldoet.'};

  return {ok:false,msg:'Je mist nog een deel van het gebied.'};

}
